CSPY7_SETTLE = 6                  # v7: entries held fully open + centred before the look-left (6 x 35 ms = 0.21 s)


def _cspy7_plan():
    """City Spy (229) v7 slot-1 timeline (10/18 user: the eye OPENS FULLY before it starts moving): the lids part with
    the iris centred, a short settle, then the v6 look-left / hold / sweep right / hold; the v6 overlapped close is kept
    (the lids start closing while the iris returns from the right).  Same tuple format as _cspy6_plan."""
    def ap(t): return max(0, min(CSPY5_NOPEN - 1, int(round(_ease4(min(1.0, max(0.0, t))) * (CSPY5_NOPEN - 1)))))
    def gi(px): return int(round((px + 4.0) / 0.25))
    NO, NL = CSPY6_NO, 17
    seq = []
    seq += [('L', 0, 0)] * 2 + [('L', 1, 0)] * 2 + [('L', 2, 0)] * 2           # faint lid line appears
    seq += [('E', ap(i / (NO - 1)), 16) for i in range(NO)]                     # lids part, iris centred
    seq += [('E', 15, 16)] * CSPY7_SETTLE                                      # fully open, settle
    seq += [('E', 15, gi(-4.0 * _ease4((i + 1) / NL))) for i in range(NL)]      # look left
    seq += [('E', 15, gi(-4.0))] * 17                                         # hold left
    seq += [('E', 15, gi(-4.0 + 8.0 * _ease4((i + 1) / 23))) for i in range(23)]   # sweep right
    seq += [('E', 15, gi(4.0))] * 17                                          # hold right
    back = [4.0 - 4.0 * _ease4((i + 1) / 14) for i in range(14)]               # back towards centre ...
    seq += [('E', 15, gi(px)) for px in back[:CSPY6_CLOSE_AT]]
    rest = back[CSPY6_CLOSE_AT:]                                              # ... while the lids already close (v6)
    for i in range(NO):
        g = rest[i] if i < len(rest) else 0.0
        seq.append(('E', ap(1 - (i + 1) / NO), gi(g)))
    seq += [('L', 2, 0)] * 2 + [('L', 1, 0)] * 2 + [('L', 0, 0)] * 2           # lid line fades away
    seq += [('B', 0, 0)] * 57                                                 # 2 s hidden
    return seq


def cityspy_eye7_timeline(base, v5base, v4base):
    """City Spy (229) v7 slot-1 list - no new images: base = the v6 mixed frames (DualDF 6420+, indexed by
    _cspy6_combos order), v5base = v5 apertures / lid-line rows, v4base = v4 blank / gaze."""
    combos = _cspy6_combos(); ids = []
    for kind, k, g in _cspy7_plan():
        if kind == 'L': ids.append(v5base + 16 + k)
        elif kind == 'B': ids.append(v4base)
        elif k == CSPY5_NOPEN - 1: ids.append(v4base + 1 + g)
        elif g == 16: ids.append(v5base + k)
        else:
            assert (k, g) in combos, "v7 needs a lid x gaze frame the v6 set does not have: %r" % ((k, g),)
            ids.append(base + combos.index((k, g)))
    return ids
