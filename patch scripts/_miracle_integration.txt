def glyph_miracle():
    """Miracle (209, Life tier 4; 2026-10-16): a fallen warrior rising again - the corpse from ANIMATE HERO (icon 86)
    stays on the ground as a faint shadow while the same figure stands up out of it in full ink, arms spread, under a
    halo, with three soft rays of light falling on it from above."""
    im = canvas(); dr = ImageDraw.Draw(im)
    body = vanilla_ink(86).crop((3, 20, 35, 35)).point(lambda v: int(v * 0.45))                     # the corpse, faded
    paste_max(im, body.resize((32 * S, 12 * S), Image.BICUBIC), (4 * S, 23 * S))
    rays = Image.new("L", (W * S, H * S), 0); rd = ImageDraw.Draw(rays)
    for x0, x1 in ((20.0, 20.0), (12.0, 15.0), (28.0, 25.0)):
        rd.polygon([((x0 - 0.9) * S, -1 * S), ((x0 + 0.9) * S, -1 * S), ((x1 + 2.2) * S, 26 * S), ((x1 - 2.2) * S, 26 * S)], fill=120)
    rays = rays.filter(ImageFilter.GaussianBlur(0.9 * S)); paste_max(im, rays)
    fig = Image.new("L", (W * S, H * S), 0); fd = ImageDraw.Draw(fig)
    def P(pts): fd.polygon([(x * S, y * S) for x, y in pts], fill=255)
    cx = 20.0
    fd.ellipse([(cx - 2.1) * S, 6.6 * S, (cx + 2.1) * S, 10.8 * S], fill=255)                          # head
    P([(cx - 2.6, 11.4), (cx + 2.6, 11.4), (cx + 2.0, 19.4), (cx + 3.4, 27.0), (cx + 1.0, 27.0), (cx, 21.0),
       (cx - 1.0, 27.0), (cx - 3.4, 27.0), (cx - 2.0, 19.4)])                                          # torso + legs
    P([(cx - 2.4, 11.6), (cx - 2.0, 13.6), (cx - 8.6, 9.0), (cx - 9.4, 6.4), (cx - 8.0, 6.2)])          # left arm raised
    P([(cx + 2.4, 11.6), (cx + 2.0, 13.6), (cx + 8.6, 9.0), (cx + 9.4, 6.4), (cx + 8.0, 6.2)])          # right arm raised
    ring = Image.new("L", (W * S, H * S), 0); stroke_ellipse(ImageDraw.Draw(ring), ((cx - 3.4) * S, 2.6 * S, (cx + 3.4) * S, 5.0 * S), int(0.8 * S), 255)

    paste_max(im, fig); paste_max(im, ring)
    return im

--- 3. build_dual_sphere_fx_v1.py ---

def miracle_sparkles():
    """Miracle (209) slot 0 (2026-10-16): Animate Dead's PARTICLE.ILB 15-17 twinkle (the 18 sparkles the corpse spawns
    at the FX stage) at 2x, inverted and tinted warm gold-white for the additive (770) mode - like Harvest Souls'
    purple_sparkles, but holy light."""
    lib = ilbcodec.read(PARTICLE); out = []
    for src in (15, 16, 17):
        im, r = raw_pixels(lib, src); im = im.convert("RGBA")
        im = im.resize((im.width * 2, im.height * 2), Image.BICUBIC)
        p = im.load()
        for y in range(im.height):
            for x in range(im.width):
                r_, g_, b_, a_ = p[x, y]; v = 1.0 - max(r_, g_, b_) / 255.0
                p[x, y] = (int(255 * min(1.0, 1.05 * v)), int(255 * min(1.0, 0.88 * v + 0.1 * v ** 3)), int(255 * min(1.0, 0.45 * v + 0.4 * v ** 3)), 255)
        out.append(im)
    return out
