def tectonic_dust(n=3):
    """Tectonic Shift (219) slots 0-2 (2026-10-17): three single frames for Air Mastery's map-wide drift (200x116,
    additive at low blend, as Ice Age's snow_drift): a faint ochre dust haze along a wide S-curve, a jagged amber
    fissure (forked crack line with a soft glow) running through it, and tan grit / pebble specks thrown up beside the
    crack."""
    W, H = 200, 116; out = []; yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    for v in range(n):
        rnd = random.Random(2190 + v); ph = v * 2.1
        acc = np.zeros((H, W, 3), np.float32)
        for k in range(12):                                                      # dust haze puffs
            t = rnd.random(); x = 10 + 180 * t; y = 60 + 26 * math.sin(t * 2 * math.pi * 0.9 + ph) + rnd.uniform(-6, 6)
            g = 0.16 * np.exp(-((xx - x) ** 2 + ((yy - y) / 0.55) ** 2) / 260.0)
            for c, cv in enumerate((200, 150, 90)): acc[..., c] += g * cv
        pts = []; x = 14.0; y = 60 + 26 * math.sin(ph) + rnd.uniform(-4, 4)       # the fissure: a jagged walk along the curve
        while x < 186:
            pts.append((x, y)); x += rnd.uniform(6, 11)
            y = 60 + 26 * math.sin((x - 10) / 180.0 * 2 * math.pi * 0.9 + ph) + rnd.uniform(-5, 5)
        glow = Image.new("L", (W, H), 0); gd = ImageDraw.Draw(glow)
        gd.line(pts, fill=255, width=5)
        for k in range(4):                                                       # side forks
            i = rnd.randrange(1, len(pts) - 1); fx, fy = pts[i]; ang = rnd.uniform(-2.6, -0.5) if k % 2 else rnd.uniform(0.5, 2.6)
            fork = [(fx, fy)]
            for s in range(3):
                fx += 7 * math.cos(ang) + rnd.uniform(-2, 2); fy += 7 * math.sin(ang) * 0.6 + rnd.uniform(-2, 2); fork.append((fx, fy))
            gd.line(fork, fill=170, width=3)
        g = np.asarray(glow.filter(ImageFilter.GaussianBlur(3.0)), np.float32) / 255.0 * 0.55
        for c, cv in enumerate((255, 150, 50)): acc[..., c] += g * cv
        im = Image.fromarray(np.clip(acc, 0, 255).astype(np.uint8), "RGB").convert("RGBA"); dr = ImageDraw.Draw(im)
        dr.line(pts, fill=(255, 190, 90, 255), width=1)                         # the bright crack core
        for k in range(40):                                                      # grit and pebbles thrown up
            t = rnd.random(); x = 10 + 180 * t; y = 60 + 26 * math.sin(t * 2 * math.pi * 0.9 + ph) + rnd.uniform(-22, 22)
            c = int(rnd.uniform(150, 230)); col = (c, int(c * 0.8), int(c * 0.55), 255)
            r = rnd.choice((0.6, 1.0, 1.4))
            if r < 1: dr.point((x, y), fill=col)
            else: dr.polygon([(x - r, y), (x, y - r), (x + r, y + 0.4), (x + 0.3, y + r)], fill=col)
        out.append(im)
    return out

    Grass 1 / Desert 2 / Snow 3 / Steppe 4 / Wasteland 5 with overlay 0xFF (bare)  -> overlay 2 HILL

        of that terrain - MOUNTAIN.ILB hills, 2 per terrain)
    same terrains with overlay 2 (hill) -> overlay 0 MOUNTAIN (single-hex mountain resource of that terrain),
       only at a "simple point"; if no mountain resource fits that terrain the hill is put back.
    Never: Water 0, Ice 6, Lava 9, Dirt 0xC, forest 1, vegetation 3, road 4, bridge 5, existing mountains.
  Underground + depths (levels 1, 2, ...):
    EarthWall 7 -> Dirt 0xC (dig; must touch an open hex; never the outer ring)  - plain cave floor, no decoration
    Dirt 0xC (overlay 0xFF) -> EarthWall 7 (bury; only at a simple point)       - rocks/bones on it are cleared
    Never: Rock 8, Lava 9, CaveWater 0xA, CaveIce 0xD, rubble/vegetation overlays.
  Rates per caster turn: changes = area x 2/1000 x (level+1) (min 1, odd-rounded), picks = 12 x that:
    64x64 map -> 9 surface / 17 underground / 25 depths;  96x96 -> 19 / 37 / 55.  (TS_PM / PICK_MULT in the script.)
  Refresh: PlaceTerrain/ClearTerrain create/free the terrain objects (art + overlay); ChangeTerrain (map VMT+0xD0)
    rewrites [field+0x14] and rebuilds the hexagons; movement costs are read live from the fields; LOS/fog for dug /
    buried Earth walls is rebuilt the same turn by build_los_terrain.py's InvalidateMap trigger in
