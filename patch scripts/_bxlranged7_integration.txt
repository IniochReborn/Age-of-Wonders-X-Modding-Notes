def skull_death_aura(n=12):
    """Hurl Skull slot 0 layer 1 of 3 (under the bone) - v7 (2026-10-19 user: 'excessive, the projectile looks
    massive'): a TIGHT smooth black death aura, MULTIPLY (1026): ~85 % dark hugging the skull (r <= 6.5 px), a soft
    Gaussian fall-off to nothing by ~10 px, a gentle breathing pulse and two short dark wisps.  Footprint ~1.7x the
    skull (was ~3.5x).  64x64, centre 31/33, 12 frames in step with the skull."""
    W = H = 64; cx, cy = 31.0, 33.0
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    out = []
    for i in range(n):
        ph = 2 * math.pi * i / n
        R = 2.6 + 0.3 * math.sin(ph)
        r = np.sqrt((xx - cx) ** 2 + ((yy - cy) * 1.05) ** 2)
        core = 0.85 * np.exp(-(np.maximum(r - 6.5, 0) / R) ** 2)
        wisp = np.zeros_like(r)
        for k, (ox, sp) in enumerate(((-2.5, 1.0), (2.5, 1.3))):
            t = ((i / n) * sp + k / 2.0) % 1.0
            wx = cx + ox + 1.0 * math.sin(ph + k * 2.1); wy = cy - 6 - 5 * t; wr = 1.8 + 1.4 * t
            wisp = np.maximum(wisp, 0.40 * (1 - t) * np.exp(-((xx - wx) ** 2 + (yy - wy) ** 2) / (wr * wr)))
        out.append(_sk5_mul(np.maximum(core, wisp)))
    return out


def skull_fire_aura(n=12):
    """Hurl Skull slot 0 layer 3 of 3 (over the bone) - v7: a THIN flickering fire rim just outside the black aura,
    ADDITIVE (770), vanilla fire colours: a ring at ~8 px (width ~1.4 px) and 10 short soft tongues (~4 px) licking
    upward, hollow inside r < 6.5 so the bone stays white in its black shroud.  Total reach ~11 px (was ~22).
    64x64, centre 31/33, 12 frames."""
    W = H = 64; cx, cy = 31.0, 33.0
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    rnd = random.Random(5712)
    tongues = [(rnd.uniform(0, 2 * math.pi), rnd.uniform(0.7, 1.3), rnd.uniform(0, 6.28)) for _ in range(10)]
    stops = np.array([s[0] for s in SK5_FIRE], np.float32)
    out = []
    for i in range(n):
        ph = 2 * math.pi * i / n
        ang = np.arctan2(yy - cy, xx - cx); r = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
        ring = np.exp(-((r - 8.0) / 1.4) ** 2) * (0.5 + 0.5 * np.sin(ang * 3 + ph * 2) ** 2)
        heat = 0.5 * ring
        for (a0, ln, p0) in tongues:
            fl = 0.65 + 0.35 * math.sin(ph * 2 + p0)
            bx, by = cx + 7.5 * math.cos(a0), cy + 7.2 * math.sin(a0)
            for s in np.linspace(0, 1, 8):
                L = 4.2 * ln * fl * s
                px_ = bx + math.cos(a0) * L * 0.3 + 0.4 * math.sin(ph + p0 + 3 * s); py_ = by + math.sin(a0) * L * 0.15 - L
                w = 1.3 * (1 - s) ** 0.7 + 0.35
                heat = np.maximum(heat, (1 - s) ** 1.2 * 0.7 * np.exp(-((xx - px_) ** 2 + (yy - py_) ** 2) / (w * w)))
        heat = heat * np.clip((r - 6.3) / 1.6, 0, 1)
        t = 1.0 - np.clip(heat, 0, 1)
        rgb = np.zeros((H, W, 3), np.float32)
        for c in range(3):
            vals = np.array([s[1][c] for s in SK5_FIRE], np.float32)
            rgb[:, :, c] = np.interp(t, stops, vals) * np.clip(heat * 1.2, 0, 1)
        out.append(_sk5_add(rgb))
    return out


def deathfire_puff_smoke(n=16):
    """Hurl Skull slot 3 layer 1 (the TRAIL) - v7 (user: 'the tail should be more of a trailing GLOW'): a faint soft
    dark-glow dab, MULTIPLY (1026) - ~38 % dark at the centre (r 4.5 px at birth, SHRINKING to ~2 px as it ages) and
    gone by frame 11 of 16.  One is dropped per tick (~6 px apart), so they melt into one smooth dark streak that
    narrows and fades behind the skull.  48x48, centre 23/23 (unchanged canvas/anchor)."""
    W = H = 48; cx, cy = 23.0, 23.0
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    rr2 = (xx - cx) ** 2 + (yy - cy) ** 2
    out = []

        amt = 0.38 * life ** 1.2 * np.exp(-rr2 / (R * R))
        out.append(_sk5_mul(amt))
    return out
