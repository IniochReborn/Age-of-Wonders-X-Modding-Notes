---- 4. build_dual_sphere_fx_v1.py: function (before def build()) + group-table row ----
              (3150, suspend_lift(), "AIsusp%02d.BMP", 770, 90, 0x0000),   # Suspend (214) slot 0: rising wind helix (96x96, additive)
def suspend_lift(n=32):
    """Suspend (214) slot 0 (2026-10-16): 96x96 additive, anchor -48/-86 like Second Wind (centre column x 48, the
    unit's feet ~y 86): pale-cyan wind ribbons spiralling UP around the unit, lifting it - three helical streaks rising
    from a ground swirl, motes carried upward, a soft updraft glow; fades out over the last third."""
    W = H = 96; cx = 48.0; yy, xx = np.mgrid[0:H, 0:W].astype(np.float32); rnd = random.Random(2140)
    motes = [(rnd.uniform(0, 6.28), rnd.uniform(10, 22), rnd.uniform(0, 1)) for _ in range(26)]
    out = []
    for i in range(n):
        t = i / (n - 1.0); fade = 1.0 if t < 0.66 else max(0.0, 1 - (t - 0.66) / 0.34); grow = min(1.0, t / 0.2)
        acc = np.zeros((H, W, 3), np.float32)
        glow = np.exp(-((xx - cx) / 16.0) ** 2) * np.clip((yy - 20) / 66.0, 0, 1) * 0.22 * grow * fade
        for c, cv in enumerate((150, 210, 255)): acc[..., c] += glow * cv
        im = Image.fromarray(np.clip(acc, 0, 255).astype(np.uint8), "RGB"); dr = ImageDraw.Draw(im)
        for k in range(3):
            pts = []
            for s in range(40):
                u = s / 39.0; y = 88 - u * 70 * grow; ph = 6.28 * (u * 1.6 + t * 1.4) + k * 2.09
                r = 20 - 8 * u; x = cx + r * math.cos(ph); y += 0.25 * r * math.sin(ph)
                pts.append((x, y, math.sin(ph)))
            for a, b in zip(pts, pts[1:]):
                lv = (0.55 + 0.45 * (a[2] > 0)) * fade
                dr.line([a[:2], b[:2]], fill=(int(170 * lv), int(225 * lv), int(255 * lv)), width=2)
        for a0, r, ph in motes:
            u = (t * 1.3 + ph) % 1.0; y = 88 - u * 76; x = cx + r * (1 - 0.4 * u) * math.cos(a0 + 6 * u)
            lv = fade * math.sin(math.pi * u)
            dr.point((x, y), fill=(int(220 * lv), int(245 * lv), int(255 * lv)))
        out.append(im.convert("RGBA"))
    return out
