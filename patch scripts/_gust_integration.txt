---- 3. build_dual_sphere_fx_v1.py: function (before def build()) + group-table row ----
              (3034, gust_wisp(), "AIgust%d.BMP", 770, 80, 0x0000),          # Gust (212) slot 0: per-strike wind wisp along the breath fan (64x64, Cold Breath's role)
def gust_wisp(n=8):
    """Gust (212) slot 0 (2026-10-16): the per-strike puff drawn along the breath fan (Cold Breath's BOLTS 90-97 role,
    64x64, centre (32, 33), ADDITIVE) - direction-neutral (the fan can point any way): five pale grey-white wind wisps
    curling round the centre and flung outward as they spin, over a faint dust haze; thinning and fading."""
    W = H = 64; cx, cy = 32.0, 33.0; yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    out = []
    for i in range(n):
        t = i / (n - 1.0); fade = (1 - t) ** 1.1
        acc = np.zeros((H, W, 3), np.float32)
        haze = np.exp(-((xx - cx) ** 2 + ((yy - cy) / 0.75) ** 2) / (40 + 260 * t)) * 0.22 * fade
        for c, cv in enumerate((205, 200, 185)): acc[..., c] += haze * cv
        im = Image.fromarray(np.clip(acc, 0, 255).astype(np.uint8), "RGB").convert("RGBA"); dr = ImageDraw.Draw(im)
        R = 5 + 20 * t
        for k in range(5):
            a0 = 2 * math.pi * k / 5 + 2.2 * t
            pts = []
            for m in range(10):
                a = a0 + math.radians(80) * m / 9.0; r = R * (0.75 + 0.35 * m / 9.0)
                pts.append((cx + r * math.cos(a), cy + 0.75 * r * math.sin(a)))
            c = int(240 * fade); dr.line(pts, fill=(c, c, int(c * 0.96), 255), width=2 if t < 0.5 else 1)
        out.append(im)
    return out
