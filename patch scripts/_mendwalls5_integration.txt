def _mend5_stick(dr, S, x, y, rot, sc, L=13.0, w=2.0):

    twig = [(1.0, -0.6), (3.6, -3.4), (4.4, -3.0), (2.2, 0.0)]
    rim = [(px + (0.7 if px > 0 else -0.7), py + (0.7 if py > 0 else -0.7)) for px, py in bar]

    dr.polygon(P(bar), fill=(112, 74, 38, 255))
