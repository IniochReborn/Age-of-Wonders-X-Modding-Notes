def glyph_staticshield():
    """Static Shield (215, Air tier 4, 2026-10-16): a heater shield whose face is split by a bold lightning bolt (cut out
    in black so the bolt reads as the shield's emblem), with jagged static arcs crackling off its rim (small zig-zag
    sparks at the shoulders and flanks, kept clear of the rim by a dark gap)."""
    im = canvas(); dr = ImageDraw.Draw(im); P = lambda pts: [(x * S, y * S) for x, y in pts]
    cx = 20.0
    def shield(inset):
        top, sh = 5.0 + inset, 9.0 + inset
        pts = [(cx - 11.0 + inset, top), (cx + 11.0 - inset, top), (cx + 11.0 - inset, 15.0)]
        for k in range(1, 13):                                                                   # curved flank to the point
            u = k / 12.0; pts.append((cx + (11.0 - inset) * (1 - u) ** 1.25, 15.0 + (16.5 - inset * 1.2) * math.sin(u * math.pi / 2)))
        pts = pts + [(2 * cx - x, y) for x, y in reversed(pts[2:-1])] + [(cx - 11.0 + inset, 15.0)]
        return pts
    for k, (x0, y0, ang, ln) in enumerate(((cx - 12.6, 7.0, 200, 6.5), (cx + 12.6, 7.0, -20, 6.5), (cx - 12.4, 18.0, 160, 5.5),
                                            (cx + 12.4, 18.0, 20, 5.5), (cx - 4.5, 3.6, 250, 4.2), (cx + 4.5, 3.6, 290, 4.2))):
        a = math.radians(ang); ux, uy = math.cos(a), -math.sin(a); px, py = -uy, ux
        zz = [(x0 + ux * ln * t + px * (1.3 if j % 2 else -1.3) * (0 < j < 4), y0 + uy * ln * t + py * (1.3 if j % 2 else -1.3) * (0 < j < 4))
              for j, t in enumerate((0.0, 0.3, 0.55, 0.8, 1.0))]
        dr.line(P(zz), fill=255, width=int(0.9 * S), joint="curve")                              # static arcs off the rim
    dr.polygon(P(shield(-1.0)), fill=0)                                                          # dark gap round the shield
    dr.polygon(P(shield(0.0)), fill=255)                                                         # the shield
    bolt = [(cx + 2.6, 7.4), (cx - 4.6, 18.4), (cx - 0.6, 18.4), (cx - 3.0, 28.6), (cx + 5.0, 15.6), (cx + 0.9, 15.6), (cx + 4.2, 7.4)]
    dr.polygon(P(bolt), fill=0)                                                                  # the lightning bolt emblem (cut out)
    return im


---- 4. build_dual_sphere_fx_v1.py: function (before def build()) + group-table row ----
              (3200, staticshield_aura(), "AIsshld%02d.BMP", 770, 100, 0x0000),   # Static Shield (215) slot 0: crackling static shell (96x96, additive)
def staticshield_aura(n=30):
    """Static Shield (215) slot 0 (2026-10-16): 96x96, feet at (48, 86) (anchor -48/-86, Second Wind's canvas), ADDITIVE,
    30 frames: a crackling electric SHELL wraps the unit - a pale-blue egg-shaped glow (centre (48, 58)) builds up while
    jagged white-violet arcs crawl over its surface and snap out to short sparks, peaks with a bright flash, then the
    shell settles and fades with a last few flickering arcs.  Arcs are re-randomised per frame (seeded -> stable art)."""
    W = H = 96; SS = 3; cx, cy, rx, ry = 48.0, 58.0, 21.0, 27.0; rnd = random.Random(2150); out = []
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    rr = np.sqrt(((xx - cx) / rx) ** 2 + ((yy - cy) / ry) ** 2)
    for i in range(n):
        t = i / (n - 1.0)
        env = min(1.0, t / 0.3) if t < 0.3 else (1.0 if t < 0.6 else max(0.0, 1 - (t - 0.6) / 0.4))
        shell = np.exp(-((rr - 1.0) / 0.10) ** 2) * 0.55 + np.exp(-(rr / 0.95) ** 6) * 0.12
        flash = math.exp(-((t - 0.32) / 0.06) ** 2)
        acc = np.zeros((H, W, 3), np.float32)
        for c, cv in enumerate((120, 170, 255)): acc[..., c] += shell * env * cv
        for c, cv in enumerate((220, 230, 255)): acc[..., c] += np.exp(-(rr / 0.7) ** 2) * flash * 0.6 * cv
        big = Image.new("L", (W * SS, H * SS), 0); d = ImageDraw.Draw(big)
        narcs = int(round((3 + 5 * env) * (0.6 + 0.4 * rnd.random())))
        for _ in range(narcs if env > 0.05 else 0):
            a0 = rnd.uniform(0, 2 * math.pi); span = rnd.uniform(0.5, 1.3); seg = 7
            pts = []
            for k in range(seg + 1):
                a = a0 + span * k / seg; j = 1.0 + rnd.uniform(-0.12, 0.12)
                pts.append(((cx + rx * j * math.cos(a)) * SS, (cy + ry * j * math.sin(a)) * SS))
            d.line(pts, fill=255, width=SS, joint="curve")
            if rnd.random() < 0.6:                                                        # a short spark snapping outward
                a = a0 + span * rnd.random(); x0, y0 = cx + rx * math.cos(a), cy + ry * math.sin(a)
                L = rnd.uniform(5, 10); px_, py_ = x0, y0; sp = [(px_ * SS, py_ * SS)]
                for k in range(3):
                    px_ += math.cos(a) * L / 3 + rnd.uniform(-2, 2); py_ += math.sin(a) * L / 3 + rnd.uniform(-2, 2); sp.append((px_ * SS, py_ * SS))
                d.line(sp, fill=200, width=max(1, SS - 1))
        arc = np.asarray(big.filter(ImageFilter.GaussianBlur(0.6 * SS)).resize((W, H), Image.BOX), np.float32) / 255.0
        glow = np.asarray(Image.fromarray((arc * 255).astype(np.uint8)).filter(ImageFilter.GaussianBlur(2.0)), np.float32) / 255.0
        lv = max(env, 0.25 if env > 0.05 else 0.0)
        for c, (cv, gv) in enumerate(((235, 140), (240, 110), (255, 255))): acc[..., c] += (arc * cv * 1.3 + glow * gv * 0.9) * lv
        out.append(Image.fromarray(np.clip(acc, 0, 255).astype(np.uint8), "RGB").convert("RGBA"))
    return out
