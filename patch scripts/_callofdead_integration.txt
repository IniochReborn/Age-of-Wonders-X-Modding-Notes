def deadcall_sparkles():
    """Call of the Dead (211) slot 0 (2026-10-16): Animate Dead's own PARTICLE.ILB 15-17 twinkle (the 18 sparkles each
    corpse spawns at the FX stage) at 2x like Harvest's, inverted for the additive (770) mode and tinted a cold,
    sickly grave-green - every corpse on the field twinkles before it rises."""
    lib = ilbcodec.read(PARTICLE); out = []
    for src in (15, 16, 17):
        im, r = raw_pixels(lib, src); im = im.convert("RGBA")
        im = im.resize((im.width * 2, im.height * 2), Image.BICUBIC)
        p = im.load()
        for y in range(im.height):
            for x in range(im.width):
                r_, g_, b_, a_ = p[x, y]; v = 1.0 - max(r_, g_, b_) / 255.0
                p[x, y] = (int(255 * min(1.0, 0.35 * v + 0.3 * v ** 3)), int(255 * min(1.0, 1.0 * v)), int(255 * min(1.0, 0.45 * v + 0.25 * v ** 3)), 255)
        out.append(im)
    return out
