SK5_DEATH = np.array([34, 22, 44], np.float32)    # vanilla-death multiply tint: near-black with a cold violet cast
SK5_FIRE = [(0.00, (255, 246, 190)), (0.25, (255, 210, 90)), (0.50, (255, 150, 35)), (0.75, (225, 80, 12)), (1.00, (120, 28, 4))]


def _sk5_firecol(t):
    """Vanilla-style fire ramp (Fire Bolt / Fire Ball palette): 0 = white-yellow core .. 1 = deep red-brown edge."""
    t = min(1.0, max(0.0, t))
    for (a, ca), (b, cb) in zip(SK5_FIRE, SK5_FIRE[1:]):
        if t <= b:
            u = (t - a) / (b - a)
            return tuple(ca[c] + (cb[c] - ca[c]) * u for c in range(3))
    return SK5_FIRE[-1][1]


def _sk5_mul(dark, tint=SK5_DEATH):
    """darkening amount 0..1 (HxW) -> a MULTIPLY sprite (drawMode 1026): white = no change; kept where > 1.5 %."""
    a = np.clip(dark, 0, 1)[:, :, None]
    rgb = 255.0 * (1 - a) + tint[None, None, :] * a
    al = np.where(a[:, :, 0] > 0.015, 255, 0)
    return Image.fromarray(np.dstack([np.clip(rgb, 0, 255), al]).astype(np.uint8), "RGBA")


def _sk5_add(rgb):
    """additive light (HxWx3, 0..255) -> an ADDITIVE sprite (drawMode 770, black key = nothing added)."""
    rgb = np.clip(rgb, 0, 255)
    al = np.where(rgb.max(2) >= 4, 255, 0)
    return Image.fromarray(np.dstack([rgb, al]).astype(np.uint8), "RGBA")


def hurl_skull5(n=12):

    mediocre'): hurl_skull4's gritty bone skull and its full 30-deg/frame spin, ALONE - no smoke, plume, crimson glow or

    S = 8; W = H = 64; cx, cy = 31.0, 33.0

    def skull(ang):
        L = Image.new("RGBA", (W * S, H * S), (0, 0, 0, 0)); d = ImageDraw.Draw(L); k = S
        def E(x0, y0, x1, y1, c): d.ellipse([(cx + x0) * k, (cy + y0) * k, (cx + x1) * k, (cy + y1) * k], fill=c)
        def P(pts, c): d.polygon([((cx + x) * k, (cy + y) * k) for x, y in pts], fill=c)
        def Ln(pts, c, w): d.line([((cx + x) * k, (cy + y) * k) for x, y in pts], fill=c, width=max(1, int(w * k)))
        rim, bone, lit, shade, blk = (14, 8, 8, 255), (200, 196, 186, 255), (236, 233, 224, 255), (124, 116, 110, 255), (4, 2, 2, 255)
        E(-5.4, -6.6, 5.4, 3.4, rim)

        E(-4.0, -2.2, -0.4, 1.5, blk); E(0.4, -2.2, 4.0, 1.5, blk)

        for t in (-1.8, -0.6, 0.6, 1.8): Ln([(t, 3.7), (t, 5.4)], blk, 0.42)
        Ln([(-1.2, -5.2), (-0.4, -3.6), (-1.0, -2.6)], rim, 0.32)
        Ln([(2.0, -4.6), (1.2, -3.4), (1.9, -2.5)], rim, 0.28)
        return L.rotate(ang, resample=Image.BICUBIC, center=(cx * k, cy * k))

    out = []
    for i in range(n):
        sk = skull(-30.0 * i)
        small = np.array(sk.resize((W, H), Image.LANCZOS)).astype(np.float32)

def skull_death_aura(n=12):
    """Hurl Skull v5 slot 0 layer 1 of 3 (drawn FIRST, under the bone): a heavy, perfectly smooth BLACK death aura -
    MULTIPLY (1026) so the darkening is a true gradient (no dither): ~92 % dark out to ~8 px round the skull, falling off as a
    soft halo to ~17 px (a clear black band between the bone and the fire), a slow breathing pulse and a few dark wisps curling up and back; cold violet-black tint like
    the vanilla Death spells.  64x64, centre 31/33, 12 frames in step with the skull."""
    W = H = 64; cx, cy = 31.0, 33.0
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    out = []
    for i in range(n):
        ph = 2 * math.pi * i / n
        R = 9.0 + 0.8 * math.sin(ph)
        r = np.sqrt((xx - cx) ** 2 + ((yy - cy) * 1.05) ** 2)
        core = 0.92 * np.exp(-(np.maximum(r - 8.0, 0) / R) ** 2 * 1.8)
        wisp = np.zeros_like(r)
        for k, (ox, sp) in enumerate(((-4.0, 1.0), (3.5, 1.3), (0.0, 0.8))):
            t = ((i / n) * sp + k / 3.0) % 1.0
            wx = cx + ox + 2.0 * math.sin(ph + k * 2.1); wy = cy - 6 - 13 * t; wr = 3.2 + 3.0 * t
            wisp = np.maximum(wisp, 0.55 * (1 - t) * np.exp(-((xx - wx) ** 2 + (yy - wy) ** 2) / (wr * wr)))
        out.append(_sk5_mul(np.maximum(core, wisp)))
    return out


def skull_fire_aura(n=12):
    """Hurl Skull v5 slot 0 layer 3 of 3 (drawn LAST, over the bone): the FIRE wrapped round the death - ADDITIVE (770),
    vanilla fire colours (white-yellow -> orange -> red-brown), a flickering ring of flame tongues standing out of the
    black aura at ~11-16 px and licking upward (~11 px), hollow inside (r < 10 adds nothing, the skull stays
    bone-white inside its black shroud).  64x64, centre 31/33, 12 frames in step with the skull."""
    W = H = 64; cx, cy = 31.0, 33.0
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    rnd = random.Random(5512)
    tongues = [(rnd.uniform(0, 2 * math.pi), rnd.uniform(0.7, 1.3), rnd.uniform(0, 6.28)) for _ in range(14)]
    stops = np.array([s[0] for s in SK5_FIRE], np.float32)
    out = []
    for i in range(n):
        ph = 2 * math.pi * i / n
        ang = np.arctan2(yy - cy, xx - cx); r = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
        ring = np.exp(-((r - 13.0) / 2.6) ** 2) * (0.45 + 0.55 * np.sin(ang * 4 + ph * 2) ** 2)
        heat = 0.55 * ring
        for (a0, ln, p0) in tongues:                                          # tongues: outward, bent upward
            fl = 0.65 + 0.35 * math.sin(ph * 2 + p0)
            bx, by = cx + 12.0 * math.cos(a0), cy + 11.5 * math.sin(a0)
            for s in np.linspace(0, 1, 12):
                L = 11.0 * ln * fl * s
                px_ = bx + math.cos(a0) * L * 0.35 + 0.8 * math.sin(ph + p0 + 3 * s); py_ = by + math.sin(a0) * L * 0.2 - L * 1.0
                w = 2.2 * (1 - s) ** 0.7 + 0.5
                heat = np.maximum(heat, (1 - s) ** 1.1 * 0.85 * np.exp(-((xx - px_) ** 2 + (yy - py_) ** 2) / (w * w)))
        heat = heat * np.clip((r - 9.5) / 3.0, 0, 1)                        # hollow inside the black band
        t = 1.0 - np.clip(heat, 0, 1)
        rgb = np.zeros((H, W, 3), np.float32)
        for c in range(3):
            vals = np.array([s[1][c] for s in SK5_FIRE], np.float32)
            rgb[:, :, c] = np.interp(t, stops, vals) * np.clip(heat * 1.25, 0, 1)
        out.append(_sk5_add(rgb))
    return out


def deathfire_puff_smoke(n=16):
    """Hurl Skull v5 slot 3 layer 1 (the TRAIL): one deathfire puff left behind every tick by
    patch_bxl_ranged_abilities_v5 (kind-0x26 helper, marker 0xCC03).  Black death smoke - MULTIPLY (1026), smooth:
    a soft violet-black blob that swells (r 7 -> 16 px), drifts up a little and thins to nothing over 16 frames, with
    a ragged edge from slow noise.  48x48, centre 23/23."""
    W = H = 48; cx, cy = 23.0, 23.0
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    noise = np.array(Image.fromarray((np.random.RandomState(5520).rand(12, 12) * 255).astype(np.uint8)).resize((W, H), Image.BICUBIC)).astype(np.float32) / 255
    out = []
    for i in range(n):
        t = i / (n - 1.0)
        R = 7.5 + 9.0 * t ** 0.7; oy = -6.0 * t
        r = np.sqrt((xx - cx) ** 2 + (yy - cy - oy) ** 2) / R
        amt = (0.82 * (1 - t) ** 1.2) * np.clip(1.25 - r - 0.35 * (noise - 0.5), 0, 1) ** 1.4
        out.append(_sk5_mul(amt))
    return out


def deathfire_puff_fire(n=16):
    """Hurl Skull v5 slot 3 layer 2 (the TRAIL): the fire inside each puff - ADDITIVE (770), vanilla fire colours: a
    hot orange core (r 5 px) that cools to red and dies within ~9 frames, a few sparks rising out of it.  48x48."""
    W = H = 48; cx, cy = 23.0, 23.0
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    rnd = random.Random(5530)
    sparks = [(rnd.uniform(-5, 5), rnd.uniform(-1, 1), rnd.uniform(0.6, 1.2)) for _ in range(3)]
    out = []
    for i in range(n):
        t = i / (n - 1.0)
        life = max(0.0, 1.0 - t / 0.55)
        rgb = np.zeros((H, W, 3), np.float32)
        if life > 0:
            R = 5.5 * (0.7 + 0.3 * life)
            heat = 0.8 * life ** 1.4 * np.exp(-((xx - cx) ** 2 + (yy - cy + 2 * t) ** 2) / (R * R))
            col = np.array(_sk5_firecol(0.25 + 0.7 * (1 - life)), np.float32)
            rgb += heat[:, :, None] * col[None, None, :]
        for (sx, s0, sp) in sparks:
            u = t * sp + 0.1 * s0
            if 0 <= u < 0.9:
                x = cx + sx * (0.4 + u); y = cy - 3 - 14 * u
                col = np.array(_sk5_firecol(0.35 + 0.65 * u), np.float32) * (1 - u) * 0.6
                rgb += np.exp(-((xx - x) ** 2 + (yy - y) ** 2) / 0.9)[:, :, None] * col[None, None, :]
        out.append(_sk5_add(rgb))
    return out


def skull_death_burst(n=30):
    """Hurl Skull v5 slot 2 layer 1 (drawn BEFORE Self Destruct's vanilla EXPLBIG 0-29, which stays untouched on top):
    the added DEATH - MULTIPLY (1026) on vanilla's 274x258 canvas / anchor (impact at 131/142): a heavy violet-black
    pall swelling under the blast (r 30 -> 125 px), a dark SHOCK RING racing out past the fireball (frames 5-20),
    ~70 black shards / smoke clots flung outward on low arcs (gravity) with soot streaks, and three black smoke
    columns boiling up from the crater late (frames 12-29); everything thins out by frame 29.  Same 30 frames /
    timing as the explosion (f18 0) - the chain's first sequence also sets the helper's lifetime (30 frames)."""
    W, H = 274, 258; cx, cy = 131.0, 142.0
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    rr = np.sqrt((xx - cx) ** 2 + ((yy - cy) * 1.3) ** 2)
    rnd = random.Random(5540)
    parts = []
    for _ in range(70):
        a = rnd.uniform(0, 2 * math.pi); v = rnd.uniform(2.4, 5.6); up = rnd.uniform(0.8, 3.0)
        parts.append((math.cos(a) * v, math.sin(a) * v * 0.55 - up, rnd.uniform(1.6, 3.4), rnd.uniform(0, 3)))
    cols = [(rnd.uniform(-22, 22), rnd.uniform(0, 4), rnd.uniform(9, 14)) for _ in range(3)]
    noise = np.array(Image.fromarray((np.random.RandomState(5541).rand(26, 28) * 255).astype(np.uint8)).resize((W, H), Image.BICUBIC)).astype(np.float32) / 255
    out = []
    for i in range(n):
        t = i / (n - 1.0)
        R = 30 + 95 * t ** 0.6
        pall = (0.80 * (1 - t) ** 1.3) * np.exp(-(rr / R) ** 2 * 1.4) * (0.8 + 0.4 * noise)
        dark = pall * min(1.0, (i + 1) / 2.0)
        if 5 <= i <= 20:                                                    # dark shock ring
            u = (i - 5) / 15.0; Rr = 70 + 70 * u
            dark = np.maximum(dark, 0.7 * (1 - u) ** 1.2 * np.exp(-((rr - Rr) / (7 + 6 * u)) ** 2) * (0.75 + 0.5 * noise))
        for (vx, vy, sz, d0) in parts:
            f = i - d0
            if f < 0: continue
            x = cx + vx * f; y = cy + vy * f + 0.11 * f * f
            fade = max(0.0, 1.0 - f / 27.0) ** 0.8
            s = sz * (1.0 + 0.05 * f)
            x0, x1 = int(max(0, x - 40)), int(min(W, x + 40)); y0, y1 = int(max(0, y - 40)), int(min(H, y + 40))
            if x0 >= x1 or y0 >= y1: continue
            sub = dark[y0:y1, x0:x1]; sx_ = xx[y0:y1, x0:x1]; sy_ = yy[y0:y1, x0:x1]
            for k in range(5):                                                 # head + trailing soot streak
                hx = x - vx * k * 0.9; hy = y - (vy + 0.22 * f) * k * 0.9
                w = s * (1 - 0.15 * k)
                sub = np.maximum(sub, (0.95 * fade * (1 - 0.18 * k)) * np.exp(-((sx_ - hx) ** 2 + (sy_ - hy) ** 2) / (w * w)))
            dark[y0:y1, x0:x1] = sub
        if i >= 12:                                                            # black smoke columns boiling up
            u = (i - 12) / 17.0
            for (ox, d0, wd) in cols:
                for k in range(6):
                    v = u - k * 0.06
                    if v <= 0: continue
                    px_ = cx + ox + 4 * math.sin(k + i * 0.3); py_ = cy - 8 - 70 * v
                    w = wd * (0.7 + 0.8 * v)
                    dark = np.maximum(dark, 0.85 * (1 - u) ** 0.8 * (1 - 0.1 * k) * np.exp(-((xx - px_) ** 2 + (yy - py_) ** 2) / (w * w)))
        out.append(_sk5_mul(dark))
    return out
