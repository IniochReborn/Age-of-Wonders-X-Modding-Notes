def glyph_mimic():
    """Mimic (233, Secret L2): the user's reference mirror frame (art import export/mimic.jpg, ornate oval with scroll
    work) TRACED -> mimic_frame_mask.png; inside the glass a spell star split down a dashed mirror axis - the left
    half solid, the right half its hollow reflection."""
    im = _user_mask_glyph("mimic_frame_mask.png")
    dr = ImageDraw.Draw(im); P = lambda pts: [(x * S, y * S) for x, y in pts]
    cx, cy, R, r = 20.0, 17.6, 7.6, 3.1
    star = []
    for k in range(10):
        a = math.radians(90 + 36 * k); rr = R if k % 2 == 0 else r
        star.append((cx + rr * math.cos(a), cy - rr * math.sin(a)))
    left = Image.new("L", im.size, 0); ImageDraw.Draw(left).polygon(P(star), fill=255)
    half = Image.new("L", im.size, 0); ImageDraw.Draw(half).rectangle([0, 0, int((cx - 0.55) * S), H * S], fill=255)
    solid = ImageChops.multiply(left, half)
    ring = Image.new("L", im.size, 0); ImageDraw.Draw(ring).polygon(P(star), fill=255)
    inner = ring.filter(ImageFilter.MinFilter(int(1.3 * S) | 1))
    outline = ImageChops.subtract(ring, inner)
    rhalf = Image.new("L", im.size, 0); ImageDraw.Draw(rhalf).rectangle([int((cx + 0.55) * S), 0, W * S, H * S], fill=255)
    outline = ImageChops.multiply(outline, rhalf)
    im.paste(255, mask=solid); im.paste(255, mask=outline)
    for y0 in (7.4, 11.0, 23.2, 26.8):                                                    # the dashed mirror axis
        dr.line(P([(cx, y0), (cx, y0 + 1.8)]), fill=255, width=int(0.8 * S))
    for x, y in ((15.2, 11.0), (25.8, 26.4)):                                              # two glints on the glass
        dr.line(P([(x - 1.2, y), (x + 1.2, y)]), fill=255, width=int(0.6 * S)); dr.line(P([(x, y - 1.2), (x, y + 1.2)]), fill=255, width=int(0.6 * S))
    return im

def mimic_shimmer(n=20):
    """Mimic (233) slot 1 (2026-10-18): the kind-0x26 driver at the CASTER's hex (patch_tc_mimic_v1.py), additive,
    64x96 (hex centre at (32, 76)): a tall oval pane of silvery-violet light forms around the caster (frames 0-5), a
    bright sheen sweeps across it left -> right and is answered by its mirror image sweeping back right -> left
    (6-13), then the pane breaks into glints that drift up and fade (14-19)."""
    W, H = 64, 96; cx, cy, A, B = 32.0, 50.0, 17.0, 30.0
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32); rnd = random.Random(2330)
    e = np.sqrt(((xx - cx) / A) ** 2 + ((yy - cy) / B) ** 2)
    glints = [(rnd.uniform(0, 6.283), rnd.uniform(0.6, 1.05), rnd.uniform(0.6, 1.4)) for _ in range(22)]
    out = []
    for i in range(n):
        t = i / (n - 1.0); acc = np.zeros((H, W, 3), np.float32)
        form = min(1.0, i / 5.0); fade = 1.0 if i < 14 else max(0.0, 1.0 - (i - 13) / 6.0)
        rim = 0.85 * np.exp(-((e - 1.0) / 0.07) ** 2) * form * fade
        pane = np.clip(1.0 - e, 0, 1) ** 0.6 * 0.22 * form * fade
        sheen = np.zeros_like(xx)
        if 6 <= i <= 13:
            k = (i - 6) / 7.0
            for pos, w in ((-1.0 + 2.0 * k, 0.9), (1.0 - 2.0 * k, 0.6)):                  # the sweep and its mirror
                u = (xx - cx) / A + 0.35 * (yy - cy) / B * (1 if w > 0.7 else -1)
                sheen += w * np.exp(-((u - pos) / 0.16) ** 2)
            sheen *= (e < 1.0)
        for c, (rc, pc, sc) in enumerate(((150, 105, 235), (170, 140, 235), (235, 220, 255))):
            acc[..., c] += rim * rc + pane * pc + sheen * sc * fade
        im = Image.fromarray(np.clip(acc, 0, 255).astype(np.uint8), "RGB").convert("RGBA"); dr = ImageDraw.Draw(im)
        if i >= 12:
            g = (i - 12) / 7.0
            for a, v, s in glints:
                x = cx + A * v * math.cos(a); y = cy + B * v * math.sin(a) - 22 * g * s
                lum = int(255 * max(0.0, 1.0 - g) ** 0.7)
                if lum < 20: continue
                col = (lum, int(lum * 0.9), 255 if lum > 0 else 0, 255); r = 1.6 if s > 1.0 else 1.0
                dr.line([(x - r, y), (x + r, y)], fill=col); dr.line([(x, y - r), (x, y + r)], fill=col)
        out.append(im)
    return out
